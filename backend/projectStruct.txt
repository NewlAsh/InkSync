collaborative-editor/
│
├── config/
│   └── db.js                 # MongoDB connection setup
│
├── models/

│   └── docRoutes.js          # /api/documents
│
├── middlewares/
│   └── authMiddleware.js     # HTTP JWT verification middleware
│
├── sockets/
│   ├── socketAuth.js         # Socket.io JWT handshake verification middleware
│   └── documentHandler.js    # Decoupled socket event listeners (join, edit, disconnect)
│
├── server.js                 # App entry point (wires Express, HTTP Server, & Socket.io)
